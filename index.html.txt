  <script>
    /**
     * Définition officielle des 12 rubriques de l'album ESCN CARDS dans l'ordre exact :
     * - A1 à A4 : Logo du club (cartes puzzle / grand format, libellé "Logo / Grand format")
     * - A5 à A9 : Infrastructures (terrain, vestiaire... -> images individuelles)
     * - B1..J4 : Photos de groupe en 3 ou 4 images (34 cartes)
     * - 1..179 : Portraits individuels (179 cartes)
     * Total : 38 cartes puzzle/photo groupe + 184 images individuelles = 222 cartes.
     */
    function buildNumberRange(start, end) {
      const list = [];
      for (let i = start; i <= end; i++) list.push(String(i));
      return list;
    }

    const ALBUM_CATEGORIES_SPEC = [
      { id: 'logo',     name: 'Logo du club',    codes: ['A1','A2','A3','A4'], mode: 'logo_puzzle' },
      { id: 'infra',    name: 'Infrastructures', codes: ['A5','A6','A7','A8','A9'], mode: 'individual_only' },
      { id: 'comites',  name: 'Comités',         codes: ['B1','B2','B3','B4', ...buildNumberRange(1, 13)], mode: 'group_and_indiv' },
      { id: 'arbitres', name: 'Arbitres',        codes: [...buildNumberRange(14, 15)], mode: 'individual_only' },
      { id: 'educ',     name: 'Éducateurs',      codes: ['C1','C2','C3', ...buildNumberRange(16, 34)], mode: 'group_and_indiv' },

      { id: 'u8u9',     name: 'U8-U9',           codes: ['E1','E2','E3','E4', ...buildNumberRange(53, 69)], mode: 'group_and_indiv' },
      { id: 'u10u11',   name: 'U10-U11',         codes: ['F1','F2','F3','F4', ...buildNumberRange(70, 88)], mode: 'group_and_indiv' },

    ];

    const ALL_CARDS = [];
    const CARD_BY_CODE = new Map();

    ALBUM_CATEGORIES_SPEC.forEach((cat, catIndex) => {
      cat.codes.forEach(code => {
        const hasLetterPrefix = /^[A-J]\d+$/.test(code);
        // Seules A1-A4 (logo_puzzle) et B1..J4 (group_and_indiv avec lettre) sont des cartes puzzle/grand format.
        // A5-A9 (Infrastructures) sont des images individuelles au même titre que 1..179.
        const isGroupOrGrand = (cat.mode === 'logo_puzzle') || (cat.mode === 'group_and_indiv' && hasLetterPrefix);
        const typeLabel = cat.mode === 'logo_puzzle'
          ? 'Logo / Grand format'
          : (isGroupOrGrand ? 'Photo groupe' : 'Image individuelle');

        const card = {
          index: ALL_CARDS.length,
          code: code,
          categoryId: cat.id,
          categoryName: cat.name,
          catIndex: catIndex,
          hasLetterPrefix: hasLetterPrefix,
          isGroupOrGrand: isGroupOrGrand,
          typeLabel: typeLabel
        };
        ALL_CARDS.push(card);
        CARD_BY_CODE.set(code.toUpperCase(), card);
      });
    });

    const TOTAL_GROUP_OR_GRAND = ALL_CARDS.filter(c => c.isGroupOrGrand).length; // 38 cartes

    const STORAGE_KEY = 'escn_cards_album_222_v2';
    const DEFAULT_TITLE = 'ESCN CARDS';

    let state = {

      customNames: {} // { [code]: string }

    let ui = {

    let packTokens = [];

    // ==================== INITIALISATION & SAUVEGARDE ====================
    function init() {
      loadState();
      bindEvents();
      renderAll();
    }

    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) {
          const parsed = JSON.parse(raw);
          state.clubName = (parsed && typeof parsed.clubName === 'string' && parsed.clubName.trim())
            ? parsed.clubName.trim()
            : DEFAULT_TITLE;
          state.counts = (parsed && typeof parsed.counts === 'object' && parsed.counts !== null)
            ? parsed.counts
            : {};
          state.customNames = (parsed && typeof parsed.customNames === 'object' && parsed.customNames !== null)
            ? parsed.customNames
            : {};
        }
      } catch (err) {
        console.error('Erreur chargement localStorage:', err);
      }
      document.getElementById('config-club-name').value = state.clubName;
    }

    function saveState() {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      } catch (err) {
        showToast('⚠️ Sauvegarde locale indisponible');
      }
    }

    // ==================== GESTION DES QUANTITÉS ====================
    function setCardQty(code, nextQty) {
      if (!CARD_BY_CODE.has(code)) return;
      const cleanQty = Math.max(0, Math.min(99, parseInt(nextQty, 10) || 0));
      if (cleanQty === 0) {
        delete state.counts[code];
      } else {
        state.counts[code] = cleanQty;
      }
      saveState();
      renderAll();
    }

    function handleCardTap(code) {
      const current = state.counts[code] || 0;
      if (current === 0) {
        setCardQty(code, 1);
      } else if (current === 1) {
        setCardQty(code, 0);
      } else {
        const doubles = current - 1;
        openConfirmModal(
          `Retirer la carte ${code} ?`,
          `Tu as cette carte possédée + ${doubles} double${doubles > 1 ? 's' : ''} (total ${current}). Veux-tu remettre sa quantité à 0, ou utiliser le bouton − pour retirer un double ?`,
          () => setCardQty(code, 0)
        );
      }
    }

    function addOneToCategory(categoryId) {
      const cat = ALBUM_CATEGORIES_SPEC.find(c => c.id === categoryId);
      if (!cat) return;
      cat.codes.forEach(code => {
        state.counts[code] = (state.counts[code] || 0) + 1;
      });
      saveState();
      renderAll();
      showToast(`✅ +1 ajouté aux ${cat.codes.length} cartes (${cat.name})`);
    }

    // ==================== STATISTIQUES ====================
    function computeStats() {
      let owned = 0;
      let duplicates = 0;
      let groupOwned = 0;

      ALL_CARDS.forEach(card => {
        const qty = state.counts[card.code] || 0;
        if (qty >= 1) {
          owned++;
          if (card.isGroupOrGrand) groupOwned++;
        }
        if (qty >= 2) {
          duplicates += (qty - 1);
        }
      });

      const missing = ALL_CARDS.length - owned;
      const percent = Math.round((owned / ALL_CARDS.length) * 100);
      return { owned, missing, duplicates, groupOwned, percent };
    }

    // ==================== SYSTÈME D'ÉCHANGES ALB222 ====================
    function encodeAlbumState222() {
      let bigVal = 0n;
      for (let i = 0; i < ALL_CARDS.length; i++) {
        const qty = state.counts[ALL_CARDS[i].code] || 0;
        const trinary = qty === 0 ? 0n : (qty === 1 ? 1n : 2n);
        bigVal = bigVal * 3n + trinary;
      }
      return 'ALB222:' + bigVal.toString(36).toUpperCase();
    }

    function decodeAlbumState222(rawText) {
      const match = String(rawText || '').match(/ALB222:([0-9A-Za-z]+)/);
      if (!match) throw new Error('Code ALB222 introuvable');

      const base36Str = match[1].toLowerCase();
      let bigVal = 0n;
      for (let i = 0; i < base36Str.length; i++) {
        const digit = parseInt(base36Str[i], 36);
        if (isNaN(digit)) throw new Error('Caractère invalide');
        bigVal = bigVal * 36n + BigInt(digit);
      }

      const states = new Array(ALL_CARDS.length).fill(0);
      for (let i = ALL_CARDS.length - 1; i >= 0; i--) {
        states[i] = Number(bigVal % 3n);
        bigVal = bigVal / 3n;
      }
      if (bigVal !== 0n) throw new Error('Code hors plage');
      return states;
    }

    function buildWhatsAppMessage() {
      const missingList = [];
      const doublesList = [];

      ALL_CARDS.forEach(card => {
        const qty = state.counts[card.code] || 0;
        if (qty === 0) {
          missingList.push(card.code);
        } else if (qty >= 2) {
          const extra = qty - 1;
          doublesList.push(extra > 1 ? `${card.code} (x${extra})` : card.code);
        }
      });

      let msg = `⚽ *${state.clubName}* (Courcelles sur Nied)\n\n`;
      msg += `❌ *Mes manquantes (${missingList.length}) :*\n`;
      msg += missingList.length > 0 ? missingList.join(', ') : 'Aucune, album complet ! 🏆';
      msg += `\n\n🔁 *Mes doubles (${doublesList.length}) :*\n`;
      msg += doublesList.length > 0 ? doublesList.join(', ') : 'Aucun double disponible.';
      msg += `\n\n📲 *Code d'échange rapide :*\n${encodeAlbumState222()}`;
      return msg;
    }

    function runTradeComparison() {
      const raw = document.getElementById('partner-code-input').value.trim();
      if (!raw) {
        showToast("Colle d'abord le code ALB222 d'un collectionneur");
        return;
      }
      try {
        const partnerStates = decodeAlbumState222(raw);
        const receiveCards = [];
        const giveCards = [];

        ALL_CARDS.forEach((card, idx) => {
          const myQty = state.counts[card.code] || 0;
          const partnerState = partnerStates[idx];
          if (myQty === 0 && partnerState === 2) receiveCards.push(card);
          if (myQty >= 2 && partnerState === 0) giveCards.push(card);
        });

        lastTradeResult = { receiveCards, giveCards };
        renderTradeComparisonResult(receiveCards, giveCards);
        showToast('🤝 Comparaison terminée !');
      } catch (err) {
        showToast('❌ Code ALB222 invalide ou incomplet');
      }
    }

    function applyValidatedTrade() {
      if (!lastTradeResult) return;

      openConfirmModal(
        'Valider cet échange ?',
        `Tu vas ajouter ${receiveCards.length} carte(s) manquante(s) et retirer 1 double sur ${giveCards.length} carte(s).`,
        () => {
          receiveCards.forEach(c => {

        previewEl.innerHTML = `<span class="text-slate-700">Saisie en cours : <strong>${escapeHtml(currentDraft)}</strong></span>`;
      } else {
        previewEl.className = 'min-h-[42px] rounded-xl px-3 py-2 bg-slate-50 border border-slate-200 flex items-center justify-between text-xs';
        previewEl.innerHTML = `<span class="text-slate-400">Aperçu instantané : tape ou compose un code (ex: <strong>A1</strong>, <strong>A5</strong> ou <strong>53</strong>)</span>`;
      }

      const listEl = document.getElementById('pack-detected-list');
      document.getElementById('pack-detected-count').textContent = String(packTokens.length);

      if (packTokens.length === 0) {
        listEl.innerHTML = `<span class="text-xs text-slate-400 italic p-1">Aucune carte ajoutée au lot pour l'instant...</span>`;
      } else {
        listEl.innerHTML = packTokens.map((code, i) => {
          const card = CARD_BY_CODE.get(code);
          return `
            <span class="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-lg text-xs font-bold ${
              card.isGroupOrGrand
                ? 'bg-escnYellow-100 text-escnBlue-950 border border-escnYellow-400'
                : 'bg-escnBlue-100 text-escnBlue-950 border border-escnBlue-300'
            }">
              <span>${code}</span>
              <button type="button" onclick="removePackToken(${i})" class="text-slate-500 hover:text-red-600 font-black ml-0.5">×</button>
            </span>
          `;
        }).join('');
      }

      const validateBtn = document.getElementById('btn-validate-pack');
      validateBtn.disabled = packTokens.length === 0;
      validateBtn.textContent = packTokens.length > 0
        ? `✅ Valider le paquet (${packTokens.length} carte${packTokens.length > 1 ? 's' : ''})`
        : 'Valider le paquet (0 carte)';
    }

    function commitDraftToPackTokens() {
      const input = document.getElementById('pack-input');
      const rawTokens = input.value.toUpperCase().split(/[\s,;]+/).filter(Boolean);
      let added = 0;

      rawTokens.forEach(tok => {
        if (CARD_BY_CODE.has(tok)) {
          packTokens.push(tok);
          added++;
        }
      });

      if (added > 0) {
        input.value = '';
        updatePackUI();
      } else if (rawTokens.length > 0) {
        showToast(`⚠️ Code inconnu : ${rawTokens.join(', ')}`);
      }
    }

    function removePackToken(idx) {
      packTokens.splice(idx, 1);
      updatePackUI();
    }

    function validateFullPack() {
      const input = document.getElementById('pack-input');
      const leftover = input.value.toUpperCase().split(/[\s,;]+/).filter(Boolean);
      leftover.forEach(tok => {
        if (CARD_BY_CODE.has(tok)) packTokens.push(tok);
      });

      if (packTokens.length === 0) return;

      let newCardsCount = 0;
      let newDoublesCount = 0;

      packTokens.forEach(code => {
        const before = state.counts[code] || 0;
        if (before === 0) newCardsCount++;
        else newDoublesCount++;
        state.counts[code] = before + 1;
      });

      saveState();
      renderAll();
      closePackModal();
      showToast(`🎉 Paquet validé : +${newCardsCount} nouvelle(s), +${newDoublesCount} double(s) !`);
    }

    // ==================== CONFIG & SAUVEGARDE ====================
    function applyBulkNames() {
      const text = document.getElementById('bulk-names-textarea').value.trim();
      if (!text) {
        showToast('Colle une liste au format Code: Nom ou Code;Nom');
        return;
      }

      let updated = 0;
      text.split('\n').forEach(line => {
        const match = line.match(/^([^:;]+)[:;](.*)$/);
        if (!match) return;
        const code = match[1].trim().toUpperCase();
        const name = match[2].trim();
        if (CARD_BY_CODE.has(code)) {
          if (name) state.customNames[code] = name;
          else delete state.customNames[code];
          updated++;
        }
      });

      saveState();
      renderAll();
      showToast(`✅ ${updated} nom(s) mis à jour !`);
    }

    // ==================== UTILITAIRES UI ====================
    function copyTextSafe(text, toastMsg) {
      const ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed';
      ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.select();
      try {
        document.execCommand('copy');
        showToast(toastMsg);
      } catch (err) {
        showToast('❌ Copie impossible');
      }
      document.body.removeChild(ta);
    }

    function showToast(msg) {
      const box = document.getElementById('toast-box');
      const el = document.createElement('div');
      el.className = 'toast-anim bg-escnBlue-950 text-white text-xs font-bold px-4 py-3 rounded-full shadow-xl border border-escnYellow-400/70 whitespace-nowrap';
      el.textContent = msg;
      box.innerHTML = '';
      box.appendChild(el);
      setTimeout(() => {
        el.style.opacity = '0';
        el.style.transition = 'opacity 0.2s ease';
        setTimeout(() => el.remove(), 200);
      }, 2500);
    }

    function openConfirmModal(title, message, onConfirm) {
      const modal = document.getElementById('confirm-modal');
      document.getElementById('confirm-modal-title').textContent = title;
      document.getElementById('confirm-modal-msg').textContent = message;
      modal.classList.remove('hidden');

      const btnOk = document.getElementById('confirm-modal-ok');
      const btnCancel = document.getElementById('confirm-modal-cancel');

      const close = () => {
        modal.classList.add('hidden');
        btnOk.onclick = null;
        btnCancel.onclick = null;
      };

      btnCancel.onclick = close;
      btnOk.onclick = () => {
        close();
        onConfirm();
      };
    }

    function escapeHtml(str) {
      const div = document.createElement('div');
      div.textContent = str;
      return div.innerHTML;
    }

            if (parsed && typeof parsed.counts === 'object' && parsed.counts !== null) {
